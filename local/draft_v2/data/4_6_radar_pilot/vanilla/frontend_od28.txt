    ctx.lineTo(padding, padding);
    ctx.stroke();

    // Draw grid lines
    const ySteps = 5;
    const yStepSize = (height - 2 * padding) / ySteps;

        ctx.moveTo(padding, y);
        ctx.lineTo(width - padding, y);
        ctx.stroke();
    }

    // Draw labels
    ctx.font = '12px Arial';
    ctx.fillStyle = '#000';
    for (let i = 0; i < ySteps; i++) {
        const y = height - padding - i * yStepSize;
        ctx.fillText((maxValue / ySteps * i).toFixed(0), padding - 10, y + 5);

    // Draw x-axis labels
    const xSteps = data.length;

}
